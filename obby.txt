local function createSnowLayer(parent, count, zIndex)
	local layer = Instance.new("Frame")
	layer.Name = "SnowLayer"
	layer.Size = UDim2.new(1, 0, 1, 0)
	layer.BackgroundTransparency = 1
	layer.ZIndex = zIndex or 11
	layer.Parent = parent

	for i = 1, count do
		local flake = Instance.new("Frame")
		local size = math.random(1, 3)
		flake.Size = UDim2.new(0, size, 0, size)
		flake.Position = UDim2.new(math.random(), 0, -0.03, 0)
		flake.BackgroundColor3 = Color3.fromRGB(255, 255, 255)
		flake.BackgroundTransparency = math.random(55, 80) / 100
		flake.BorderSizePixel = 0
		flake.ZIndex = zIndex or 11

		local corner = Instance.new("UICorner")
		corner.CornerRadius = UDim.new(1, 0)
		corner.Parent = flake
		flake.Parent = layer

		local speed = math.random(18, 40) / 10
		local drift = math.random(-25, 25) / 10

		task.spawn(function()
			while flake and flake.Parent do
				local startX = flake.Position.X.Scale
				local tween = TweenService:Create(flake, TweenInfo.new(speed, Enum.EasingStyle.Linear), {
					Position = UDim2.new(startX + (drift / 100), 0, 1.03, 0)
				})
				tween:Play()
				tween.Completed:Wait()
				if flake and flake.Parent then
					flake.Position = UDim2.new(math.random(), 0, -0.03, 0)
				end
			end
		end)
	end
	return layer
end

local function createScanlines(parent)
	local scan = Instance.new("Frame")
	scan.Name = "Scanlines"
	scan.Size = UDim2.new(1, 0, 1, 0)
	scan.BackgroundTransparency = 1
	scan.ZIndex = 100
	scan.Parent = parent

	local pattern = Instance.new("UIGradient")
	pattern.Color = ColorSequence.new({
		ColorSequenceKeypoint.new(0, Color3.fromRGB(255, 255, 255)),
		ColorSequenceKeypoint.new(0.5, Color3.fromRGB(255, 255, 255)),
		ColorSequenceKeypoint.new(0.5, Color3.fromRGB(0, 0, 0)),
		ColorSequenceKeypoint.new(1, Color3.fromRGB(0, 0, 0))
	})
	pattern.Transparency = NumberSequence.new({
		NumberSequenceKeypoint.new(0, 0.993),
		NumberSequenceKeypoint.new(1, 0.993)
	})
	pattern.Offset = Vector2.new(0, 0)
	pattern.Rotation = 0
	pattern.Parent = scan

	task.spawn(function()
		while scan and scan.Parent do
			pattern.Offset = Vector2.new(0, (tick() * 0.4) % 1)
			task.wait(0.04)
		end
	end)

	return scan
end

	content.Parent = self.contentArea

	local contentList = Instance.new("UIListLayout")
	contentList.SortOrder = Enum.SortOrder.LayoutOrder
	contentList.Padding = UDim.new(0, 8)
	contentList.Parent = content

	local contentPad = Instance.new("UIPadding")
	contentPad.PaddingTop = UDim.new(0, 4)
	contentPad.PaddingBottom = UDim.new(0, 10)

	self.pages[name] = {
		button = tabBtn,
		content = content,
		elements = {},
		accent = accentLine,
		stroke = tabStroke
	}

	tabBtn.MouseButton1Click:Connect(function()
		self:SwitchPage(name)
	end)

	tabBtn.MouseEnter:Connect(function()
		if self.activePage ~= name then
			TweenService:Create(tabBtn, TweenInfo.new(0.2), {BackgroundTransparency = 0.94}):Play()
			TweenService:Create(tabStroke, TweenInfo.new(0.2), {Transparency = 0.92}):Play()
		end
	end)

	tabBtn.MouseLeave:Connect(function()
		if self.activePage ~= name then
			TweenService:Create(tabBtn, TweenInfo.new(0.2), {BackgroundTransparency = 0.97}):Play()
			TweenService:Create(tabStroke, TweenInfo.new(0.2), {Transparency = 0.97}):Play()
		end
	end)

	if not self.activePage then
		self:SwitchPage(name)
	end

	return self.pages[name]
end

function GhostHub:SwitchPage(name)
	local page = self.pages[name]
	if not page then return end

	if self.searchBox.Text ~= "" then
		self.searchBox.Text = ""
		self:UpdateSearch("")
	end

	if self.activePage and self.pages[self.activePage] then
		local cur = self.pages[self.activePage]
		cur.content.Visible = false
		TweenService:Create(cur.button, TweenInfo.new(0.2), {
			BackgroundTransparency = 0.97,
			TextTransparency = 0.55
		}):Play()
		TweenService:Create(cur.stroke, TweenInfo.new(0.2), {Transparency = 0.97}):Play()
		TweenService:Create(cur.accent, TweenInfo.new(0.2), {Size = UDim2.new(0, 3, 0, 0)}):Play()
	end

	self.activePage = name
	page.content.Visible = true
	TweenService:Create(page.button, TweenInfo.new(0.2), {
		BackgroundTransparency = 0.92,
		TextTransparency = 0.15
	}):Play()
	TweenService:Create(page.stroke, TweenInfo.new(0.2), {Transparency = 0.88}):Play()
	TweenService:Create(page.accent, TweenInfo.new(0.25, Enum.EasingStyle.Back), {
		Size = UDim2.new(0, 3, 0, 18)
	}):Play()
end

function GhostHub:LayoutPage(name)
	local page = self.pages[name]
	if not page then return end
	local yPos = 0
	for _, info in ipairs(page.elements) do
		if info.element and info.element.Parent == page.content then
			info.element.Position = UDim2.new(0, 0, 0, yPos)
			yPos = yPos + info.height
		end
	end
	page.content.CanvasSize = UDim2.new(0, 0, 0, yPos + 20)
end

function GhostHub:UpdateSearch(query)
	query = query:lower():gsub("^%s*(.-)%s*$", "%1")
	if query == "" then
		self.searchFrame.Visible = false
		if self.activePage then
			self.pages[self.activePage].content.Visible = true
		end
		for _, info in ipairs(self.allElements) do
			if info.element and info.originalParent then
				info.element.Parent = info.originalParent
			end
		end
		for name, _ in pairs(self.pages) do
			self:LayoutPage(name)
		end
		return
	end

	for _, p in pairs(self.pages) do
		p.content.Visible = false
	end
	self.searchFrame.Visible = true

	for _, child in pairs(self.searchFrame:GetChildren()) do
		if child:IsA("GuiObject") then
			child:Destroy()
		end
	end

	local yPos = 0
	for _, info in ipairs(self.allElements) do
		if info.name:lower():find(query, 1, true) then
			info.element.Parent = self.searchFrame
			info.element.Position = UDim2.new(0, 0, 0, yPos)
			yPos = yPos + info.height
		end
	end

	self.searchFrame.CanvasSize = UDim2.new(0, 0, 0, yPos + 20)
end

function GhostHub:_ensurePage(pageName)
	if not self.pages[pageName] then
		self:AddPage(pageName)
	end
	return self.pages[pageName]
end

-- CHECKPOINT SİSTEMİ MANTIĞI VE HUB ENTEGRASYONU

local checkpointsFolder = workspace:WaitForChild("Checkpoints")
local TP_DELAY = 0.5
local isTeleporting = false
local currentCPIndex = 1

local function teleportToPart(part)
	local character = player.Character or player.CharacterAdded:Wait()
	local hrp = character:FindFirstChild("HumanoidRootPart")
	if hrp and part then
		if part:IsA("BasePart") then
			hrp.CFrame = part.CFrame + Vector3.new(0, 3, 0)
		elseif part:IsA("Model") then
			hrp.CFrame = part:GetPivot() + Vector3.new(0, 3, 0)
		end
	end
end

local function getSortedCheckpoints()
	local cpList = {}
	for _, child in ipairs(checkpointsFolder:GetChildren()) do
		local numberName = tonumber(child.Name)
		if numberName then
			table.insert(cpList, {part = child, num = numberName})
		end
	end
	table.sort(cpList, function(a, b) return a.num < b.num end)
	return cpList
end

local function startTeleportingLoop()
	local sortedCPs = getSortedCheckpoints()
	
	for i = currentCPIndex, #sortedCPs do
		if not isTeleporting then break end
		currentCPIndex = i
		teleportToPart(sortedCPs[i].part)
		task.wait(TP_DELAY)
	end
	
	if currentCPIndex >= #sortedCPs then
		currentCPIndex = 1
	end
end

local hub = GhostHub.new("GHOSTHUB", "CHECKPOINT TELEPORTER")

hub:AddLabel("Main", "Işınlanma Ayarları")

-- 1. Checkpoint Git/Arama Kutusu
hub:AddTextBox("Main", "CP No Yaz ve Enter'a Bas (Örn: 15)...", function(text, enterPressed)
	if enterPressed and text ~= "" then
		local targetNum = tonumber(text)
		if targetNum then
			local sortedCPs = getSortedCheckpoints()
			local foundIndex = nil
			
			for index, item in ipairs(sortedCPs) do
				if item.num == targetNum then
					foundIndex = index
					break
				end
			end
			
			if foundIndex then
				currentCPIndex = foundIndex
				teleportToPart(sortedCPs[foundIndex].part)
			end
		end
	end
end)

-- 2. Otomatik TP Aç/Kapa Toggle
hub:AddToggle("Main", "Oto-TP Başlat", false, function(state)
	isTeleporting = state
	if isTeleporting then
		task.spawn(startTeleportingLoop)
	end
end)
